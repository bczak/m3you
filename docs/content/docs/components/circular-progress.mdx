import { CircularProgress } from 'm3you';

Use determinate progress when you know how much work is left, and indeterminate
when you do not. The `wavy` variant is the Material 3 Expressive treatment — the
track ripples rather than sitting flat.

<Playground
  name="CircularProgress"
  controls={{
    type: ['determinate', 'indeterminate'],
    variant: ['flat', 'wavy'],
    size: ['md', 'sm', 'lg'],
    value: { min: 0, max: 100, default: 65 },
    strokeWidth: { min: 2, max: 12, default: 4 },
  }}
  render={(props) => <CircularProgress {...props} />}
/>
